# EyeLive: acquisition / detection / review / training / trigger application. One directory per module; every module's CMakeLists.txt is a single call of eyelive_add_module (below), so a task never has to edit a shared
# build file. Tests are every tests/test_*.cpp of the module: one ctest test each, named eyelive_<module>_<file>, labelled <module> and eyelive; they receive the path of the cpp/ directory as argument 1.
set(EYELIVE_INCLUDE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_library(eyelive_contracts INTERFACE)       # the frozen header contracts (include/eyelive/*.hpp); every module links it
target_include_directories(eyelive_contracts INTERFACE ${EYELIVE_INCLUDE})
find_package(Threads REQUIRED)

function(eyelive_add_module name)
  file(GLOB_RECURSE SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
  if(SRC)
    add_library(eyelive_${name} STATIC ${SRC})
    target_include_directories(eyelive_${name} PUBLIC ${EYELIVE_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_link_libraries(eyelive_${name} PUBLIC eyelive_contracts Threads::Threads ${ARGN})
    target_compile_options(eyelive_${name} PRIVATE -Wall -Wextra)
  else()
    add_library(eyelive_${name} INTERFACE)
    target_include_directories(eyelive_${name} INTERFACE ${EYELIVE_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_link_libraries(eyelive_${name} INTERFACE eyelive_contracts Threads::Threads ${ARGN})
  endif()
  file(GLOB TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.cpp)
  foreach(t ${TESTS})
    get_filename_component(tn ${t} NAME_WE)
    add_executable(eyelive_${name}_${tn} ${t})
    target_link_libraries(eyelive_${name}_${tn} PRIVATE eyelive_${name})
    target_compile_options(eyelive_${name}_${tn} PRIVATE -Wall -Wextra)
    add_test(NAME eyelive_${name}_${tn} COMMAND eyelive_${name}_${tn} ${CMAKE_SOURCE_DIR})
    set_tests_properties(eyelive_${name}_${tn} PROPERTIES LABELS "eyelive;${name}" TIMEOUT 600)
  endforeach()
endfunction()
